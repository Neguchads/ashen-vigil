`spikes/visual-forest/spike_main.gd`:

```gdscript
extends Node3D
## Teste visual descartável: clareira de floresta à noite com névoa, lua e vela.
## A cena é montada por código para não depender de trabalho manual no editor.

const TREE_SCENE := "res://assets/tree.glb"
const GROUND_ALBEDO := "res://assets/ground_albedo.jpg"
const GROUND_NORMAL := "res://assets/ground_normal.jpg"
const GROUND_ROUGH := "res://assets/ground_rough.jpg"
const AMBIENT := "res://assets/ambient.ogg"
const LAUGH := "res://assets/laugh.ogg"
const TREE_COUNT := 600
const AREA_SIZE := 120.0
const CLEARING_RADIUS := 10.0

var _candle: OmniLight3D
var _laugh: AudioStreamPlayer3D
var _time := 0.0
var _next_laugh := 8.0


func _ready() -> void:
	_build_environment()
	_build_moon()
	_build_ground()
	_scatter_trees()
	_build_candle()
	_build_audio()
	var camera := Camera3D.new()
	camera.set_script(load("res://fps_probe.gd"))
	add_child(camera)
	print("Terrain3D disponível: ", ClassDB.class_exists("Terrain3D"))


func _process(delta: float) -> void:
	_time += delta
	# Chama da vela: duas senoides fora de fase imitam o tremor.
	_candle.light_energy = 1.4 + sin(_time * 13.0) * 0.12 + sin(_time * 7.3) * 0.08
	if _laugh != null and _time >= _next_laugh:
		_laugh.position = Vector3(randf_range(-30, 30), 1.0, randf_range(-30, 30))
		_laugh.play()
		_next_laugh = _time + randf_range(18.0, 30.0)


func _build_environment() -> void:
	var sky_mat := ProceduralSkyMaterial.new()
	sky_mat.sky_top_color = Color(0.02, 0.03, 0.06)
	sky_mat.sky_horizon_color = Color(0.05, 0.06, 0.09)
	sky_mat.ground_bottom_color = Color(0.01, 0.01, 0.02)
	sky_mat.ground_horizon_color = Color(0.05, 0.06, 0.09)
	var sky := Sky.new()
	sky.sky_material = sky_mat

	var env := Environment.new()
	env.background_mode = Environment.BG_SKY
	env.sky = sky
	env.ambient_light_source = Environment.AMBIENT_SOURCE_SKY
	env.ambient_light_energy = 0.15
	env.tonemap_mode = Environment.TONE_MAPPER_ACES
	env.sdfgi_enabled = true
	env.ssao_enabled = true
	env.glow_enabled = true
	env.volumetric_fog_enabled = true
	env.volumetric_fog_density = 0.035
	env.volumetric_fog_albedo = Color(0.6, 0.65, 0.75)

	var world_env := WorldEnvironment.new()
	world_env.environment = env
	add_child(world_env)


func _build_moon() -> void:
	var moon := DirectionalLight3D.new()
	moon.rotation_degrees = Vector3(-35.0, 40.0, 0.0)
	moon.light_color = Color(0.6, 0.7, 1.0)
	moon.light_energy = 0.35
	moon.light_volumetric_fog_energy = 1.5
	moon.shadow_enabled = true
	add_child(moon)


func _build_ground() -> void:
	var mat := StandardMaterial3D.new()
	mat.albedo_texture = load(GROUND_ALBEDO)
	mat.normal_enabled = true
	mat.normal_texture = load(GROUND_NORMAL)
	mat.roughness_texture = load(GROUND_ROUGH)
	mat.uv1_scale = Vector3(30.0, 30.0, 1.0)
	var plane := PlaneMesh.new()
	plane.size = Vector2(AREA_SIZE, AREA_SIZE)
	var ground := MeshInstance3D.new()
	ground.mesh = plane
	ground.material_override = mat
	add_child(ground)


func _scatter_trees() -> void:
	var tree_root: Node3D = (load(TREE_SCENE) as PackedScene).instantiate()
	var rng := RandomNumberGenerator.new()
	rng.seed = 42
	var transforms: Array[Transform3D] = []
	while transforms.size() < TREE_COUNT:
		var pos := Vector3(
			rng.randf_range(-AREA_SIZE / 2, AREA_SIZE / 2),
			0.0,
			rng.randf_range(-AREA_SIZE / 2, AREA_SIZE / 2)
		)
		if pos.length() < CLEARING_RADIUS:
			continue
		var basis := Basis(Vector3.UP, rng.randf() * TAU).scaled(Vector3.ONE * rng.randf_range(0.8, 1.4))
		transforms.append(Transform3D(basis, pos))
	# Uma MultiMesh por malha do modelo (tronco, folhas...), todas com as mesmas posições.
	for mesh_node in tree_root.find_children("*", "MeshInstance3D", true, false):
		var mi := mesh_node as MeshInstance3D
		var local := tree_root.global_transform.affine_inverse() * mi.global_transform
		var mm := MultiMesh.new()
		mm.transform_format = MultiMesh.TRANSFORM_3D
		mm.mesh = mi.mesh
		mm.instance_count = transforms.size()
		for i in transforms.size():
			mm.set_instance_transform(i, transforms[i] * local)
		var mmi := MultiMeshInstance3D.new()
		mmi.multimesh = mm
		add_child(mmi)
	tree_root.free()


func _build_candle() -> void:
	_candle = OmniLight3D.new()
	_candle.position = Vector3(0.0, 1.0, -2.0)
	_candle.light_color = Color(1.0, 0.6, 0.3)
	_candle.omni_range = 6.0
	_candle.shadow_enabled = true
	add_child(_candle)


func _build_audio() -> void:
	if ResourceLoader.exists(AMBIENT):
		var music := AudioStreamPlayer.new()
		music.stream = load(AMBIENT)
		music.volume_db = -6.0
		add_child(music)
		music.play()
	if ResourceLoader.exists(LAUGH):
		_laugh = AudioStreamPlayer3D.new()
		_laugh.stream = load(LAUGH)
		_laugh.unit_size = 6.0
		add_child(_laugh)
```

`spikes/visual-forest/fps_probe.gd`:

```gdscript
extends Camera3D
## Modo padrão: percorre a clareira por 20 s, mede fps e tira 3 capturas.
## Modo --explore: voo livre (WASD, mouse, Shift corre, Esc sai).

const DURATION := 20.0
const RADIUS := 25.0
const SPEED := 6.0

var _t := 0.0
var _samples: PackedFloat32Array = []
var _shots: Array[float] = [5.0, 10.0, 15.0]
var _explore := false


func _ready() -> void:
	current = true
	_explore = "--explore" in OS.get_cmdline_user_args()
	position = Vector3(0.0, 1.7, 8.0)
	if _explore:
		Input.mouse_mode = Input.MOUSE_MODE_CAPTURED


func _unhandled_input(event: InputEvent) -> void:
	if not _explore:
		return
	if event is InputEventMouseMotion:
		rotation.y -= event.relative.x * 0.002
		rotation.x = clampf(rotation.x - event.relative.y * 0.002, -1.4, 1.4)
	elif event is InputEventKey and event.pressed and event.keycode == KEY_ESCAPE:
		get_tree().quit()


func _process(delta: float) -> void:
	if _explore:
		_fly(delta)
		return
	_t += delta
	position = Vector3(cos(_t * 0.3) * RADIUS, 1.7, sin(_t * 0.3) * RADIUS)
	look_at(Vector3(0.0, 1.5, 0.0))
	if _t > 1.0:  # o primeiro segundo é carregamento
		_samples.append(1.0 / delta)
	if not _shots.is_empty() and _t >= _shots[0]:
		_shots.pop_front()
		get_viewport().get_texture().get_image().save_png("user://spike_shot_%d.png" % int(_t))
	if _t >= DURATION:
		_finish()


func _fly(delta: float) -> void:
	var dir := Vector3.ZERO
	if Input.is_key_pressed(KEY_W):
		dir -= basis.z
	if Input.is_key_pressed(KEY_S):
		dir += basis.z
	if Input.is_key_pressed(KEY_A):
		dir -= basis.x
	if Input.is_key_pressed(KEY_D):
		dir += basis.x
	var speed := SPEED * (3.0 if Input.is_key_pressed(KEY_SHIFT) else 1.0)
	position += dir.normalized() * speed * delta


func _finish() -> void:
	var sorted := _samples.duplicate()
	sorted.sort()
	var total := 0.0
	for s in _samples:
		total += s
	var avg := total / _samples.size()
	var low := sorted[int(sorted.size() * 0.01)]
	var text := "fps_medio=%.1f fps_1pct=%.1f amostras=%d" % [avg, low, _samples.size()]
	print(text)
	var f := FileAccess.open("user://spike_fps.txt", FileAccess.WRITE)
	f.store_line(text)
	f.close()
	get_tree().quit()
```

`game/src/autoload/content_registry.gd`:

```gdscript
class_name ContentRegistryNode
extends Node
## Registro central de conteúdo: o jogo procura tudo por ID, nunca por caminho.
## O autoload se chama ContentRegistry; a classe tem outro nome para não colidir com ele.

## Substituições feitas por mods, no formato "id: mod_antigo -> mod_novo".
var overrides: Array[String] = []

var _defs: Dictionary = {}  # String -> ContentDef
var _sources: Dictionary = {}  # String -> String (mod que registrou)


func register(def: ContentDef, source_mod: String) -> bool:
	if def == null:
		push_warning("ContentRegistry: definição nula ignorada (%s)" % source_mod)
		return false
	var key := String(def.id)
	if not ContentDef.is_valid_id(key):
		push_warning("ContentRegistry: id inválido '%s' em %s" % [key, source_mod])
		return false
	if _defs.has(key):
		overrides.append("%s: %s -> %s" % [key, _sources[key], source_mod])
	_defs[key] = def
	_sources[key] = source_mod
	return true


func has_def(id: String) -> bool:
	return _defs.has(id)


func get_def(id: String) -> ContentDef:
	return _defs.get(id)


func get_source(id: String) -> String:
	return _sources.get(id, "")


func list(content_type: StringName) -> Array[ContentDef]:
	var result: Array[ContentDef] = []
	for def: ContentDef in _defs.values():
		if def.get_content_type() == content_type:
			result.append(def)
	result.sort_custom(_by_id)
	return result


static func _by_id(a: ContentDef, b: ContentDef) -> bool:
	return String(a.id) < String(b.id)


func clear() -> void:
	_defs.clear()
	_sources.clear()
	overrides.clear()
```

```gdscript
class_name SemVer
extends RefCounted
## Versões MAJOR.MINOR.PATCH e restrições simples: ">=X.Y.Z" ou "X.Y.Z" (exata).

static var _regex: RegEx


static func _parse(text: String) -> PackedInt32Array:
	if _regex == null:
		_regex = RegEx.create_from_string("^(\\d+)\\.(\\d+)\\.(\\d+)$")
	var m := _regex.search(text.strip_edges())
	if m == null:
		return PackedInt32Array()
	return PackedInt32Array([m.get_string(1).to_int(), m.get_string(2).to_int(), m.get_string(3).to_int()])


static func is_valid(text: String) -> bool:
	return _parse(text).size() == 3


## Exige versões válidas (conferir com is_valid antes).
static func compare(a: String, b: String) -> int:
	var va := _parse(a)
	var vb := _parse(b)
	for i in 3:
		if va[i] != vb[i]:
			return -1 if va[i] < vb[i] else 1
	return 0


static func is_valid_constraint(constraint: String) -> bool:
	var c := constraint.strip_edges()
	if c.begins_with(">="):
		return is_valid(c.substr(2))
	return is_valid(c)


static func satisfies(version: String, constraint: String) -> bool:
	if not is_valid(version) or not is_valid_constraint(constraint):
		return false
	var c := constraint.strip_edges()
	if c.begins_with(">="):
		return compare(version, c.substr(2)) >= 0
	return compare(version, c) == 0
```

`game/src/modding/mod_manifest.gd`:

```gdscript
class_name ModManifest
extends RefCounted
## Conteúdo validado de um mod.json. Se algo estiver errado, `error` explica o quê.

const MOD_ID_PATTERN := "^[a-z0-9_]+$"

var id := ""
var display_name := ""
var version := ""
var game_version := ""
var load_after := PackedStringArray()
var content := PackedStringArray()
## Vazio quando o manifesto é válido.
var error := ""


static func parse(text: String, folder_name: String) -> ModManifest:
	var m := ModManifest.new()
	var json := JSON.new()
	if json.parse(text) != OK:
		m.error = "mod.json inválido (linha %d): %s" % [json.get_error_line(), json.get_error_message()]
		return m
	if typeof(json.data) != TYPE_DICTIONARY:
		m.error = "mod.json precisa ser um objeto { ... }"
		return m
	var data: Dictionary = json.data
	for key: String in ["id", "name", "version", "game_version"]:
		if typeof(data.get(key)) != TYPE_STRING:
			m.error = "campo '%s' ausente ou não é texto" % key
			return m
	m.id = data["id"]
	m.display_name = data["name"]
	m.version = data["version"]
	m.game_version = data["game_version"]

	if RegEx.create_from_string(MOD_ID_PATTERN).search(m.id) == null:
		m.error = "id '%s' inválido: use só letras minúsculas, números e _" % m.id
	elif m.id != folder_name:
		m.error = "id '%s' diferente do nome da pasta '%s'" % [m.id, folder_name]
	elif not SemVer.is_valid(m.version):
		m.error = "version '%s' inválida: use X.Y.Z" % m.version
	elif not SemVer.is_valid_constraint(m.game_version):
		m.error = "game_version '%s' inválida: use >=X.Y.Z ou X.Y.Z" % m.game_version
	if m.error != "":
		return m

	var after: Variant = _string_list(data.get("load_after", []))
	if after == null:
		m.error = "load_after precisa ser uma lista de textos"
		return m
	m.load_after = after

	var paths: Variant = _string_list(data.get("content", []))
	if paths == null:
		m.error = "content precisa ser uma lista de textos"
		return m
	for path: String in paths:
		if not path.begins_with("res://"):
			m.error = "content: '%s' precisa começar com res://" % path
			return m
	m.content = paths
	return m


## Devolve PackedStringArray ou null se `value` não for uma lista só de textos.
static func _string_list(value: Variant) -> Variant:
	if typeof(value) != TYPE_ARRAY:
		return null
	var out := PackedStringArray()
	for item: Variant in value:
		if typeof(item) != TYPE_STRING:
			return null
		out.append(item)
	return out
```

```gdscript
class_name ModOrder
extends RefCounted
## Ordena mods por load_after. Dependência ausente ou ciclo desativa o mod (e quem depende dele).

const CORE_ID := "core"


class Result:
	var order: Array[ModManifest] = []
	var disabled: Dictionary = {}  # id -> motivo


static func sort(manifests: Array[ModManifest]) -> Result:
	var result := Result.new()
	var active: Dictionary = {}
	for m in manifests:
		active[m.id] = m

	# 1. Dependência ausente, repetido até estabilizar (cascata).
	var changed := true
	while changed:
		changed = false
		for id: String in active.keys():
			var m: ModManifest = active[id]
			for dep in m.load_after:
				if not active.has(dep):
					result.disabled[id] = "depende de '%s', que não está ativo" % dep
					active.erase(id)
					changed = true
					break

	# 2. Ordenação topológica em lotes; cada lote em ordem alfabética, com core na frente.
	var remaining: Dictionary = active.duplicate()
	while not remaining.is_empty():
		var ready: Array[String] = []
		for id: String in remaining.keys():
			var waiting := false
			for dep in (remaining[id] as ModManifest).load_after:
				if remaining.has(dep):
					waiting = true
					break
			if not waiting:
				ready.append(id)
		if ready.is_empty():
			for id: String in remaining.keys():
				result.disabled[id] = "ciclo em load_after (ou depende de um mod em ciclo)"
			break
		ready.sort()
		if ready.has(CORE_ID):
			ready.erase(CORE_ID)
			ready.push_front(CORE_ID)
		for id in ready:
			result.order.append(remaining[id])
			remaining.erase(id)
	return result
```

```gdscript
class_name ModPacker
extends RefCounted
## Empacota a pasta content/ de um mod em <id>.pck, montado em res://mods/<id>/.
## Usa PCKPacker, que não exige templates de exportação.


static func pack(content_dir: String, mod_id: String, out_pck: String) -> Error:
	var files := _collect_files(content_dir, "")
	if files.is_empty():
		push_error("ModPacker: nenhum arquivo em %s" % content_dir)
		return ERR_FILE_NOT_FOUND
	DirAccess.make_dir_recursive_absolute(out_pck.get_base_dir())
	var packer := PCKPacker.new()
	var err := packer.pck_start(ProjectSettings.globalize_path(out_pck))
	if err != OK:
		return err
	for rel in files:
		var source := ProjectSettings.globalize_path(content_dir.path_join(rel))
		err = packer.add_file("res://mods/%s/%s" % [mod_id, rel], source)
		if err != OK:
			return err
	return packer.flush()


static func _collect_files(root: String, prefix: String) -> PackedStringArray:
	var out := PackedStringArray()
	var here := root if prefix == "" else root.path_join(prefix)
	for f in DirAccess.get_files_at(here):
		if f.ends_with(".import") or f.ends_with(".uid"):
			continue
		out.append(f if prefix == "" else prefix.path_join(f))
	for d in DirAccess.get_directories_at(here):
		out.append_array(_collect_files(root, d if prefix == "" else prefix.path_join(d)))
	return out
```

```gdscript
extends Node
## Carrega o "mod zero" (core) e os mods do jogador, nessa ordem.
## Mod com problema é desativado com motivo; o jogo nunca fecha por causa de mod.

const CORE_DIR := "res://core"
const USER_MODS_DIR := "user://mods"
const CORE_ID := "core"


class Report:
	var loaded: Array[String] = []
	var disabled: Dictionary = {}  # pasta ou id -> motivo
	var warnings: Array[String] = []


var last_report: Report


func _ready() -> void:
	last_report = load_all(ContentRegistry, CORE_DIR, USER_MODS_DIR)
	for w in last_report.warnings:
		push_warning(w)
	for id: String in last_report.disabled:
		push_warning("Mod '%s' desativado: %s" % [id, last_report.disabled[id]])


func load_all(registry: ContentRegistryNode, core_dir: String, mods_dir: String) -> Report:
	var report := Report.new()
	var game_version: String = ProjectSettings.get_setting("application/config/version", "0.0.0")

	var core := _read_manifest(core_dir.path_join("mod.json"), CORE_ID)
	if core.error != "":
		report.disabled[CORE_ID] = core.error
		report.warnings.append("core inválido: " + core.error)
		return report
	var manifests: Array[ModManifest] = [core]

	if not DirAccess.dir_exists_absolute(mods_dir):
		DirAccess.make_dir_recursive_absolute(mods_dir)
	for folder in DirAccess.get_directories_at(mods_dir):
		var m := _read_manifest(mods_dir.path_join(folder).path_join("mod.json"), folder)
		if m.error == "" and m.id == CORE_ID:
			m.error = "o id 'core' é reservado ao jogo base"
		if m.error == "" and not SemVer.satisfies(game_version, m.game_version):
			m.error = "exige jogo %s; versão atual %s" % [m.game_version, game_version]
		if m.error != "":
			report.disabled[folder] = m.error
			continue
		manifests.append(m)

	var sorted := ModOrder.sort(manifests)
	report.disabled.merge(sorted.disabled)
	for m in sorted.order:
		var failed_dep := _first_missing_dependency(m, report.loaded)
		if failed_dep != "":
			report.disabled[m.id] = "depende de '%s', que falhou ao carregar" % failed_dep
			continue
		if m.id != CORE_ID and not _mount(m, mods_dir, report):
			continue
		_register_content(registry, m, report)
		report.loaded.append(m.id)
	return report


func _read_manifest(path: String, folder: String) -> ModManifest:
	if not FileAccess.file_exists(path):
		var missing := ModManifest.new()
		missing.error = "mod.json ausente"
		return missing
	return ModManifest.parse(FileAccess.get_file_as_string(path), folder)


func _first_missing_dependency(m: ModManifest, loaded: Array[String]) -> String:
	for dep in m.load_after:
		if not loaded.has(dep):
			return dep
	return ""


func _mount(m: ModManifest, mods_dir: String, report: Report) -> bool:
	var pck := mods_dir.path_join(m.id).path_join(m.id + ".pck")
	if not FileAccess.file_exists(pck):
		report.disabled[m.id] = "arquivo %s.pck ausente" % m.id
		return false
	if not ProjectSettings.load_resource_pack(ProjectSettings.globalize_path(pck), true):
		report.disabled[m.id] = "falha ao montar %s.pck" % m.id
		return false
	return true


func _register_content(registry: ContentRegistryNode, m: ModManifest, report: Report) -> void:
	for path in m.content:
		if not ResourceLoader.exists(path):
			report.warnings.append("%s: arquivo %s não existe" % [m.id, path])
			continue
		var res := ResourceLoader.load(path)
		if not (res is ContentDef):
			report.warnings.append("%s: %s não é uma definição de conteúdo" % [m.id, path])
			continue
		if not registry.register(res, m.id):
			report.warnings.append("%s: id inválido em %s" % [m.id, path])
```

```gdscript
extends SceneTree
## Uso: godot --headless --path game -s res://tools/build_mod.gd -- <pasta_do_mod> <pasta_de_saida>
## Gera <saida>/<id>/mod.json e <saida>/<id>/<id>.pck.


func _init() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size() != 2:
		printerr("Uso: -- <pasta_do_mod> <pasta_de_saida>")
		quit(2)
		return
	var src: String = args[0]
	var out_root: String = args[1]
	var manifest := ModManifest.parse(FileAccess.get_file_as_string(src.path_join("mod.json")), src.get_file())
	if manifest.error != "":
		printerr("mod.json inválido: ", manifest.error)
		quit(1)
		return
	var out_dir := out_root.path_join(manifest.id)
	DirAccess.make_dir_recursive_absolute(out_dir)
	var err := ModPacker.pack(src.path_join("content"), manifest.id, out_dir.path_join(manifest.id + ".pck"))
	if err != OK:
		printerr("Falha ao empacotar: ", error_string(err))
		quit(1)
		return
	DirAccess.copy_absolute(src.path_join("mod.json"), out_dir.path_join("mod.json"))
	print("Mod gerado em ", out_dir)
	quit(0)
```

```powershell
# Empacota mods/<Mod> em build/mods/<Mod>/. Com -Install, copia para a pasta de mods do jogo.

    [switch]$Install
)

$godot = if ($env:GODOT_BIN) { $env:GODOT_BIN } else { 'A:\AI\Tools\Godot\4.7.2\Godot_v4.7.2-stable_win64_console.exe' }

$out = Join-Path $root 'build\mods'
if (-not (Test-Path (Join-Path $src 'mod.json'))) { Write-Host "Mod não encontrado: $src" -ForegroundColor Red; exit 1 }

# O Godot espera barras normais nos caminhos.
& $godot --headless --path (Join-Path $root 'game') -s res://tools/build_mod.gd -- $src.Replace('\', '/') $out.Replace('\', '/')

if ($Install) {
    $dest = Join-Path $env:APPDATA "Godot\app_userdata\Ashen Vigil\mods\$Mod"

    Copy-Item -Force (Join-Path $out "$Mod\*") $dest

}
```
